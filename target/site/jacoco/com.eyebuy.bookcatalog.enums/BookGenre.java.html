package com.eyebuy.bookcatalog.enums;

import lombok.Getter;

@Getter
public enum BookGenre {

    FICTION("Fiction", "虚构小说"),
    NON_FICTION("Non-Fiction", "非虚构"),
    SCIENCE_FICTION("Science Fiction", "科幻"),
    FANTASY("Fantasy", "奇幻"),
    MYSTERY("Mystery", "悬疑推理"),
    ROMANCE("Romance", "言情"),
    HISTORY("History", "历史"),
    BIOGRAPHY("Biography", "传记"),
    SCIENCE("Science", "科学"),
    TECHNOLOGY("Technology", "科技"),
    BUSINESS("Business", "商业"),
    SELF_HELP("Self-Help", "自助"),
    CHILDREN("Children", "儿童读物"),
    COOKING("Cooking", "烹饪"),
    TRAVEL("Travel", "旅行"),
    ART("Art", "艺术"),
    MUSIC("Music", "音乐"),
    RELIGION("Religion", "宗教"),
    PHILOSOPHY("Philosophy", "哲学"),
    POETRY("Poetry", "诗歌");

    private final String englishName;
    private final String chineseName;

    BookGenre(String englishName, String chineseName) {
        this.englishName = englishName;
        this.chineseName = chineseName;
    }

    public static BookGenre fromString(String value) {
        if (value == null || value.isBlank()) {
            return null;
        }
        for (BookGenre genre : values()) {
            if (genre.name().equalsIgnoreCase(value)
                    || genre.englishName.equalsIgnoreCase(value)
                    || genre.chineseName.equals(value)) {
                return genre;
            }
        }
        return null;
    }
}
